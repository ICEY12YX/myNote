<mark style="background: #ABF7F7A6;">在进行转移操作后，被转移的对象就不能继续使用了，所以对象移动一般都是对临时对象进行操作（因为临时对象很快就要销毁了）</mark>

    其他情况,对临时对象进行移动语义,就要之后多积累了

移动语义的标准写法
    ![[Pasted image 20250611174531.png]]

```cpp
class Test {
public:
	Test() = default;

	Test(const Test& test) {  //拷贝构造
		if (test.str) {
			str = new char[strlen(test.str) + 1];  //strlen是不会统计结束符'\0'的,所以分配空间的时候要手动加上
			strcpy_s(str, strlen(test.str) + 1, test.str);
		}
		else {
			str = nullptr;
		}

		std::cout << "拷贝构造" << std::endl;
	}

	Test(Test&& test) {  //移动构造(不用分配内存)
		if (test.str) {
			str = test.str;
			test.str = nullptr;  //上面传进来的test不能加上const,不然这里就没法修改为nullptr
		}
		else {
			str = nullptr;
		}

		std::cout << "移动构造" << std::endl;
	}

	Test& operator=(const Test& test) {  //普通重载赋值运算符
		//自赋值情况
		if (this == &test) {  //this是指针,所以这样比较或者*this == test
			return *this;
		}

		//接下来把test.str赋值给str
		if (str) {
			delete[] str;
			str = nullptr;
		}

		if (test.str) {
			str = new char[strlen(test.str) + 1];
			strcpy_s(str, strlen(test.str) + 1, test.str);
		}
		else {  //让这个if的逻辑完整,但是其实没必要,因为上面已经让str = nullptr;了
			str = nullptr;
		}

		std::cout << "普通赋值" << std::endl;

		return *this;
	}

	Test& operator=(Test&& test) {  //移动赋值运算符
		if (this == &test) {
			return *this;
		}

		if (str) {
			delete[] str;
			str = nullptr;
		}

		//关键这里不一样,这里就是让str接管test.str
		if (test.str) {
			str = test.str;
			test.str = nullptr;
		}
		else {
			str = nullptr;
		}

		std::cout << "移动赋值" << std::endl;

		return *this;
	}

private:
	char* str = nullptr;
};

Test makeTest() {
	Test t;
	return t;
}
```

#include <iostream>

int main() {

}

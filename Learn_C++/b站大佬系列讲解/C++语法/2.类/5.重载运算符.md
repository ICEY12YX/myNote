<mark style="background: #ABF7F7A6;">“operator 运算符”来定义运算符的行为，operator是一个关键字，告诉编译器我要重载运算符了</mark>
    可以把operator++整个当做函数名,其他和普通函数其实是一样的

下面就是对各个运算符进行重载的演示
++  --
```cpp
class Test {
public:
	void operator++() {  //重载++运算符
		++count;
	}

	void operator--() {  //重载--运算符
		--count;
	}

	unsigned count = 0; //偷懒的写法
};


int main() {
	Test test;
	++test;
	std::cout << test.count << std::endl;
	--test;
	std::cout << test.count << std::endl;
	std::cin.get();
}
```

( )  重载这个就是实现仿函数(此时类对象可以当做函数使用，因此称为仿函数)
```cpp
class Test {
public:
	void operator++() {  //重载++运算符
		++count;
	}

	void operator--() {  //重载--运算符
		--count;
	}

    int operator[](unsigned i) const{  //重载[]
        //这里最好加上判断,i>0 && i<ivec.size
        return ivec[i];
    }

    void operator()() const {  //重载() 这个就是之前没讲的仿函数
        std::cout << "hello" << std::endl;
    }

    void operator()(const std::string& str) const{  //可以重载(这里可以重载,因为可以传入参数,像++这种没法传参,那重载不来)
        std::cout << str << std::endl;
    }

	unsigned count = 0; //偷懒的写法
	std::vector<int> ivec{1,2,3,4,5};
};


int main() {
	Test test;
	test();
	test("fanfanfan");
	std::cin.get();
}
```

输入输出<<  >>
```cpp
class Test {

    friend std::ostream& operator<< (std::ostream& os,const Test& test);  //输出
    friend std::istream& operator>> (std::istream& is,Test& test);  //输入

public:
	void operator++() {  //重载++运算符
		++count;
	}

	void operator--() {  //重载--运算符
		--count;
	}

    int operator[](unsigned i) const{  //重载[]
        //这里最好加上判断,i>0 && i<ivec.size
        return ivec[i];
    }

    void operator()() const {  //重载() 这个就是之前没讲的仿函数
        std::cout << "hello" << std::endl;
    }

    void operator()(const std::string& str) const{  //可以重载(这里可以重载,因为可以传入参数,像++这种没法传参,那重载不来)
        std::cout << str << std::endl;
    }

	unsigned count = 0; //偷懒的写法
	std::vector<int> ivec{1,2,3,4,5};
	std::string name;
};

std::ostream& operator<< (std::ostream& os,const Test& test){
    os << test.name << std::endl;
    return os
}

std::istream& operator>> (std::istream& is,Test& test){
    is >> test.name;
    return is
}


int main() {
	Test test;
	std::cin >> test;  //重载以后输入某个字符串,那就直接赋值给test.name
	std::cout << test;  //重载以后直接输出test.name
	std::cin.get();
}
```

```cpp
class Test {
```

```cpp
class Test {

public:
	Test& operator= (const Test& test){  //这里写返回Test&,因为下面我们用的时候是用到旧的实例
    	if(this == &test){
        	return *this;  //发生自赋值的时候,就直接返回
    	}
    	count = test.count;
    	name = test.name;
    	return *this;  //注意这种写法 返回Test&,同样是return *this
	}

	unsigned count = 0; //偷懒的写法
	std::string name;
};

int main() {
	Test test;
	Test test2;
	test2.count=20;
	test = test2;
	std::cout << test.count << std::endl;
	std::cin.get();
}
```

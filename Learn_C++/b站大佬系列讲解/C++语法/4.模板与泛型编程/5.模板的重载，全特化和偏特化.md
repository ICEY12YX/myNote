重载代码:
```cpp
//函数模版
template<typename T>
void test(const T& parm) {  //模版重载
	std::cout << "void test(const T& parm)" << std::endl;
}

template<typename T>
void test(T* parm) {   //模版重载
	std::cout << "void test(T* parm)" << std::endl;
}

void test(double parm) {  //函数重载
	std::cout << "void test(T* parm)" << std::endl;
}

int main() {
	test(200);

	int i = 100;
	test(&i);

	test(1.1);

}
```

___

模板的全特化：就是指<mark style="background: #ABF7F7A6;">模板的实参列表与与相应的模板参数列表一一对应</mark>
模板的偏特化：<mark style="background: #ABF7F7A6;">偏特化就是介于普通模板和全特化之间，只存在部分类型明确化，而非将模板唯一化</mark>
(别管概念,直接看代码)
```cpp
//类模版
template<typename T1, typename T2>
class Test {
public:
	Test() {
		std::cout << "common template" << std::endl;
	}
};

//偏特化
template<typename T1, typename T2>
class Test<T1*, T2*> {
public:
	Test() {
		std::cout << "pointer template" << std::endl;
	}
};

//偏特化
template<typename T2>
class Test<int, T2*> {
public:
	Test() {
		std::cout << "int-pointer template" << std::endl;
	}
};

//全特化
template<>
class Test<int, double> {
public:
	Test() {
		std::cout << "int-double template" << std::endl;
	}
};


int main() {
	Test<int*, int*> test;  //pointer template
	Test<int, double*> test2;  //int-pointer template
	Test<int, double> test3;  //int-double template

    //这个跟上面任何一种特化版本都不匹配,所以用到一般模版的版本
	Test<double, double> test4;  //common template

}
```

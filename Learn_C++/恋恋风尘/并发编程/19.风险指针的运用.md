上节无锁栈存在一个问题，如果每次pop都会被多个线程调用，则要删除的节点就会一直往待删除列表中增加，导致待删除列表无法被回收。

上一节,我们是规定只有一个(或者说最后一个)线程pop的时候,才去回收待删除列表
这一节,我们可以用风险指针
将要删除的节点做特殊处理，如果有线程使用它，就将他标记为正在使用，那么这个节点的指针就是风险指针，也就是不能被其他线程删除。

```cpp

std::shared_ptr<T> pop()
{
    //get_hazard_pointer_for_current_thread,查找数组中的节点，返回一个可用的节点
    //就是查找风险指针数组,返回一个空闲的没指向任何node的一个风险指针
    //这里获得了空闲的风险指针的引用(这样,这里对hp进行修改,风险指针数组里也会改)
    std::atomic<void*>& hp=get_hazard_pointer_for_current_thread(); 
    
    node* old_head=head.load();   // 取出栈里的头结点
    node* temp; 
    do {
        temp = old_head;
        hp.store(old_head);   // 把旧的头结点存入风险指针
        old_head = head.load();  
    } while(old_head != temp);   
    // ...
}
```
为什么要在do里再`old_head = head.load();`
    假设执行完`hp.store(old_head);`  head被别的线程修改了,那此时肯定不能再pop当前的old_head
    所以要`old_head = head.load();`
    并且,如果old_head发生了修改,那上一句的`hp.store(old_head);`也不对了,应该存入最新的old_head
    所以要循环判断`old_head != temp`
    不等于就再循环一次,然后old_head就是最新的值了,然后再执行一次`hp.store(old_head);`就覆盖了hp上个循环的旧值

完整的pop
```cpp
std::shared_ptr<T> pop()
{
    //get_hazard_pointer_for_current_thread,查找数组中的节点，返回一个可用的节点
    //就是查找风险指针数组,返回一个空闲的没指向任何node的一个风险指针
    //这里获得了空闲的风险指针的引用(这样,这里对hp进行修改,风险指针数组里也会改)
    std::atomic<void*>& hp=get_hazard_pointer_for_current_thread(); 
    
    node* old_head=head.load();   // 取出栈里的头结点
    do{
        node* temp;
        do{
            temp = old_head;
            hp.store(old_head);
            old_head = head.load();
        } while(old_head != temp); 
        //如果old_head和temp不等说明head被其他线程更新了，需重试
        
    }while(old_head&&!head.compare_exchange_strong(old_head,old_head->next)); 
    //但是有可能退出上一个循环后,马上head又被修改了
    //所以要head和old_head比较,等于  那就head更新为old_head->next
    //不等于  old_head=head  返回false(非一下true)
    //并且old_head不为空(true)  两个条件都是true,就继续循环
    //old_head是空,那就直接退出循环

    // 上面执行顺利的话,head已经被更新为old_head->next
    // 这里就直接将上面的风险指针清空
    hp.store(nullptr);    
    std::shared_ptr<T> res;
    if(old_head)  //old_head非空
    {
        res.swap(old_head->data);//拿到数据
        
        //删除旧有的头节点之前，先核查它是否正被风险指针所指向
        if(outstanding_hazard_pointers_for(old_head))    
        {
            //有风险指针指着old_head,就延迟删除
            reclaim_later(old_head);    
        }
        else
        {
            //没有,就直接删除
            delete old_head;    
        }
        //同时 执行到这里删除没有风险的节点
        delete_nodes_with_no_hazards();    
    }
    return res; 
    //如果上面old_head是空,直接退出循环
    //那上面if就会跳过
    //然后这里直接返回空的智能指针

}
```
为什么可以直接`hp.store(nullptr); `
    上面我们循环的时候, 把old_head放进风险指针
    是因为old_head我们还要用,要用到old_head->next
    所以还不希望old_head被删掉,所以用风险指针保护着
    而head更新完以后,我们希望的是能够删掉old_head
    所以首先把这边的风险指针清空
    但是有可能别的线程找了个空闲的风险指针 存了同一个old_head
    所以后面删之前要判断一下,并做不同操作

从全局风险指针数组中返回一个空闲的节点

```cpp
class hp_owner {
public:
    hp_owner(hp_owner const&) = delete;
    hp_owner operator=(hp_owner const&) = delete;
    
    hp_owner():hp(nullptr){
        //遍历风险指针数组(因为循环max_hazard_pointers次)
        //去找一个空闲的风险指针
        for (unsigned i = 0; i < max_hazard_pointers; ++i) {
            std::thread::id old_id;  //先搞个id类型的old_id(现在是空的)

            //风险指针中的id和old_id比较
            //等于 说明这个风险指针是空的是空闲的,还没有线程要它
            //那就让这个风险指针的id为当前线程
            if (hazard_pointers[i].id.compare_exchange_strong
                (old_id, std::this_thread::get_id())) {
                hp = &hazard_pointers[i];  //拿到这个风险指针
                break;
            }
        }

        //没找到空闲的,报异常
        if (!hp) {
            throw std::runtime_error("No hazard pointers available");
        }
    }

    //析构的时候,只将风险指针的数据清空
    ~hp_owner() {
        hp->pointer.store(nullptr); //pointer指向空
        hp->id.store(std::thread::id());  //给一个空的id
    }

    std::atomic<void*>& get_pointer() {
		return hp->pointer;  //就直接返回风险指针指向的东西
	}
	
private:
    //hazard_pointer是风险指针的结构体名
    hazard_pointer* hp;
};
```

```cpp
void delete_nodes_with_no_hazards() {
    //拿到待删列表的头
    data_to_reclaim* current = nodes_to_reclaim.exchange(nullptr);
    while (current) {
        data_to_reclaim* const next = current->next; //先存下next
        //current的data没有被风险指针指着,就删掉
        if (!outstanding_hazard_pointers_for(current->data)) {
                delete current;
        }
        else {
            //current的data有风险指针指,那就加回去
            //不会重复加,因为上面nodes_to_reclaim已经变成空了
            add_to_reclaim_list(current);
        }
        current = next; //current往后移动
    }
}
```
待删列表里的元素会不会重复
    有可能的
    ![[Pasted image 20250731170126.png]]
    但是好像并不会
    因为多线程执行pop
    到`while(old_head&&!head.compare_exchange_strong(old_head,old_head->next)); `的时候
    多个线程old_head一样,但是此时应该只有一个线程会成功退出循环,其他线程就进入下一次循环,那别的线程之后的old_head就不一样了
    所以应该不会同时两个线程都跳出循环,然后对同一个old_head加入待删列表


完整代码
```cpp
﻿#pragma once
#include <condition_variable>
#include <mutex>
#include <queue>
#include <atomic>
#include <memory>

//最大风险指针数量
unsigned const max_hazard_pointers = 100;

//风险指针
struct hazard_pointer {
	std::atomic<std::thread::id> id;
	std::atomic<void*>  pointer;
};

//风险指针数组
extern hazard_pointer hazard_pointers[max_hazard_pointers];

//风险指针持有类
class hp_owner {
public:
	hp_owner(hp_owner const&) = delete;
	hp_owner operator=(hp_owner const&) = delete;
	hp_owner():hp(nullptr){
		bind_hazard_pointer();
	}

	std::atomic<void*>& get_pointer() {
		return hp->pointer;
	}

	~hp_owner() {
		hp->pointer.store(nullptr);
		hp->id.store(std::thread::id());
	}
private:

    //这个函数不用管
	void bind_hazard_pointer() {
		for (unsigned i = 0; i < max_hazard_pointers; ++i) {
			std::thread::id  old_id;
			if (hazard_pointers[i].id.compare_exchange_strong(old_id, std::this_thread::get_id())) {
				hp = &hazard_pointers[i];
				break;
			}
		}

		if (!hp) {
			throw std::runtime_error("No hazard pointers available");
		}
	}
	hazard_pointer* hp;
};

std::atomic<void*>& get_hazard_pointer_for_current_thread() {
	//每个线程都具有自己的风险指针 线程本地变量
	thread_local static hp_owner hazzard;  
	return hazzard.get_pointer();
}

template<typename T>
class hazard_pointer_stack {
private:
	
	//栈节点
	struct node {
		std::shared_ptr<T> data;
		node* next;
		node(T const& data_) :data(std::make_shared<T>(data_)) {}
	};

	//待删节点
	struct data_to_reclaim {
		node* data;
		data_to_reclaim* next;
		data_to_reclaim(node * p):data(p), next(nullptr){}
		~data_to_reclaim() {
			delete data;
		}
	};

	hazard_pointer_stack(const hazard_pointer_stack&) = delete;
	hazard_pointer_stack& operator = (const hazard_pointer_stack&) = delete;
	std::atomic<node*> head;
	std::atomic<data_to_reclaim*>  nodes_to_reclaim;
public:
	hazard_pointer_stack() {}

    //push和上一节没区别
	void push(T const& data) {
		node* const new_node = new node(data);    		
		new_node->next = head.load();    	
		while (!head.compare_exchange_weak(new_node->next, new_node));    
	}

	bool outstanding_hazard_pointers_for(void* p)
	{
		for (unsigned i = 0; i < max_hazard_pointers; ++i)
		{
			if (hazard_pointers[i].pointer.load() == p)
			{
				return true;
			}
		}
		return false;
	}

	void add_to_reclaim_list(data_to_reclaim* reclaim_node) {
		reclaim_node->next = nodes_to_reclaim.load();
		while (!nodes_to_reclaim.compare_exchange_weak(reclaim_node->next, reclaim_node));
	}

	void reclaim_later(node* old_head) {
		add_to_reclaim_list(new data_to_reclaim(old_head));
	}

	void delete_nodes_with_no_hazards() {
		data_to_reclaim* current = nodes_to_reclaim.exchange(nullptr);
			while (current) {
				data_to_reclaim* const next = current->next;
				if (!outstanding_hazard_pointers_for(current->data)) {
					delete current;
				}
				else {
					add_to_reclaim_list(current);
				}

				current = next;
			}
	}

	std::shared_ptr<T> pop() {
		//1 从风险列表中获取一个节点给当前线程
		std::atomic<void*>& hp = get_hazard_pointer_for_current_thread();
		node* old_head = head.load();
		do
		{
			node* temp;
			do
			{
				temp = old_head;
				hp.store(old_head);
				old_head = head.load();
			}//2 如果old_head和temp不等说明head被其他线程更新了，需重试
			while (old_head != temp);
		}//3 将当前head更新为old_head->next，如不满足则重试
		while (old_head &&
			!head.compare_exchange_strong(old_head, old_head->next));
		// 4一旦更新了head指针，便将风险指针清零
		hp.store(nullptr);
		std::shared_ptr<T> res;
		if (old_head)
		{
			res.swap(old_head->data);
			//5 删除旧有的头节点之前，先核查它是否正被风险指针所指涉
			if (outstanding_hazard_pointers_for(old_head))
			{
				//6 延迟删除
				reclaim_later(old_head);
			}
			else
			{
				//7 删除头部节点
				delete old_head;
			}
			//8 删除没有风险的节点
			delete_nodes_with_no_hazards();
		}
		return res;
	}

};
```

```cpp
void TestHazardPointer() {
    hazard_pointer_stack<int> hazard_stack;
    std::set<int>  rmv_set;
    std::mutex set_mtx;

    std::thread t1([&]() {
        for (int i = 0; i < 20000; i++) {
            hazard_stack.push(i);
            std::cout << "push data " << i << " success!" << std::endl;
            std::this_thread::sleep_for(std::chrono::milliseconds(5));
        }
        });

    std::thread t2([&]() {
        for (int i = 0; i < 10000;) {
            auto head = hazard_stack.pop();
            if (!head) {
                std::this_thread::sleep_for(std::chrono::milliseconds(10));
                continue;
            }
            std::lock_guard<std::mutex> lock(set_mtx);
            rmv_set.insert(*head);
            std::cout << "pop data " << *head << " success!" << std::endl;
            i++;
        }
        });

    std::thread t3([&]() {
        for (int i = 0; i < 10000;) {
            auto head = hazard_stack.pop();
            if (!head) {
                std::this_thread::sleep_for(std::chrono::milliseconds(10));
                continue;
            }
            std::lock_guard<std::mutex> lock(set_mtx);
            rmv_set.insert(*head);
            std::cout << "pop data " << *head << " success!" << std::endl;
            i++;
        }
        });

    t1.join();
    t2.join();
    t3.join();

    assert(rmv_set.size() == 20000);
}
```

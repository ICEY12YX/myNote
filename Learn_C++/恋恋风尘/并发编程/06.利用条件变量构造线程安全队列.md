# 不良实现

试想有一个线程A一直输出1，另一个线程B一直输出2。
我们想让两个线程交替输出1，2，1，2...的效果
可以这样实现
```cpp
int num = 1;
std::mutex mtx_num;

void PoorImpleman() {
    std::thread t1([]() {
        for (;;) {  //代表没有条件的循环

            {
                std::lock_guard<std::mutex> lock(mtx_num);  //修改num之前上锁
                if (num == 1) {  //如果num=1,再下面把num改成2
                    std::cout << "thread A print 1....." << std::endl;
                    num++;
                    continue;  //修改了以后,跳过下面的睡觉,进入下一次循环
                }
            }

            //如果num=2,就是上一次也是这个线程修改的num,那就这次不修改num,睡一会
            std::this_thread::sleep_for(std::chrono::milliseconds(500));

        }

     });

    std::thread t2([]() {
        for (;;) {

            {
                std::lock_guard<std::mutex> lock(mtx_num);
                if (num == 2) {  
                    std::cout << "thread B print 2....." << std::endl;
                    num--;
                    continue;
                }
            }

            std::this_thread::sleep_for(std::chrono::milliseconds(500));

        }

        });

    t1.join();
    t2.join();
}
```
但是上面的睡一会的操作不好
因为线程A检测`num`值，如果`num`不为1，则线程A就睡了，
在线程A睡眠这段时间很可能B已经处理完打印了，此时A还在睡眠(500毫秒还没结束)，是对资源的浪费，也错过了最佳的处理时机

# 条件变量

上面那样不好,所以用 条件变量
条件变量 是线程之前通信的一种方式
```cpp
int num = 1;
std::mutex mtx_num;
std::condition_variable cvA; //条件变量
std::condition_variable cvB;

void ResonableImplemention() {
    std::thread t1([]() {
        for (;;) {
            //为了配合条件变量,需要用到unique_lock
            std::unique_lock<std::mutex> lock(mtx_num);
            cvA.wait(lock, []() {  
                return num == 1;
                });  //这里前面的lock不是固定写lock
                //是前面我们的unique_lock变量叫lock

            num++;
            std::cout << "thread A print 1....." << std::endl;
            cvB.notify_one();
            //cvB.notify_all();  也可以通知所以线程
            //这么做的好处就是线程交替处理非常及时。比起`sleep`的方式，
        }

        });

    std::thread t2([]() {
        for (;;) {

            std::unique_lock<std::mutex> lock(mtx_num);
            cvB.wait(lock, []() {
                return num == 2;
                });

            num--;
            std::cout << "thread B print 2....." << std::endl;
            cvA.notify_one();
        }

        });

    t1.join();
    t2.join();
}
```

# 线程安全队列

(操作和栈差不多)
```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
template<typename T>
class threadsafe_queue {
private:
    std::mutex mut;  
    std::queue<T> data_queue;  //真正的队列数据
    std::condition_variable data_cond;  //条件变量
public:
    threadsafe_queue() {}
    threadsafe_queue(threadsafe_queue const& other){  //拷贝构造
        std::lock_guard<std::mutex> lk(other.mut);
        data_queue = other.data_queue;
    }
    
    void push(T new_value)
    {
        std::lock_guard<std::mutex> lk(mut);  //上锁
        data_queue.push(new_value);  
        //因为别的线程消费的时候,可能队列为空就挂起了
        //所以这里重新生产以后,就要notify别的线程
        data_cond.notify_one();
    }
    void wait_and_pop(T& value)  //传进来引用,下面就不用返回了
    {
        std::unique_lock<std::mutex> lk(mut);  //上锁
        //先判断队列是否为空
        //队列为空,就挂起
        //不为空继续执行(注意类里判断实例中的成员变量,要加上[this],就是lambda一样捕获对象)
        //(因为没有传进去的东西,所以[]和{}之间省略了())
        data_cond.wait(lk,[this](){return !data_queue.empty();});
        value = data_queue.front();
        data_queue.pop();
    }

    //也可以写成这个版本(和 线程安全的栈一样的写法)
    //用shared_ptr是防止作用域结束,指针被回收,导致外面得到的指针是悬空的
    //下面调用还是用的上面的版本
    std::shared_ptr<T> wait_and_pop()
    {
        std::unique_lock<std::mutex> lk(mut);  //上锁
        //先判断队列是否为空
        //队列为空,就挂起
        //不为空继续执行(注意类里判断实例中的成员变量,要加上[this],就是lambda一样捕获对象)
        //(因为没有传进去的东西,所以[]和{}之间省略了())
        data_cond.wait(lk,[this](){return !data_queue.empty();});
        std::shared_ptr<T> res(std::make_shared<T>(data_queue.front()));
        data_queue.pop();
        return res
    }

    //这个和前面wait_and_pop的区别就是,返回了bool,并且没有用条件变量阻塞
    bool try_pop(T& value)
    {
        std::lock_guard<std::mutex> lk(mut);  //加锁,不让别的push pop操作扰乱下面判断
        if(data_queue.empty()){  
            return false;  //如果为空返回false,就不pop了
        }

        //不为空就消费数据
        value = data_queue.front();
        data_queue.pop();
        return true;  //返回true
    }

    //try_pop也可以用shared_ptr写
};
```

调用
```cpp
void test_safe_que() {
    threadsafe_queue<int> safe_que;
    std::mutex mtx_print;  //写锁
    std::thread producer(  //线程producer,负责给队列push数据
        [&]() {
            for (int i = 0; ;i++) {
                safe_que.push(i);  //内部也有修改队列的锁
                {
                    //为什么这里加锁
                    //为了下面cout打印的时候不会受其他线程干扰
                    //几个线程同时打印,可能打印出来的东西就乱了
                    //这个锁不影响对队列的实际push,pop操作
                    std::lock_guard<std::mutex> printlk(mtx_print);  //上锁
                    std::cout << "producer push data is " << i << std::endl;
                }

                std::this_thread::sleep_for(std::chrono::milliseconds(200));
            }
        }
    );

    std::thread consumer1(  //消费者1,负责pop数据
        [&]() {
            for (;;) {
                int data;
                safe_que.wait_and_pop(data);
                {
                    std::lock_guard<std::mutex> printlk(mtx_print); //上锁
                    std::cout << "consumer1 wait and pop data is " << data << std::endl;
                }

                std::this_thread::sleep_for(std::chrono::milliseconds(500));
            }
        }    
    );

    std::thread consumer2(  //消费者2,负责pop数据
        [&]() {
            for (;;) {
                int data;
                bool flag = safe_que.try_pop(data);
                if (flag) {  //返回true
                    {
                        std::lock_guard<std::mutex> printlk(mtx_print); //上锁
                        std::cout << "consumer2 try_pop data is " << data << std::endl;
                    }

                }

                std::this_thread::sleep_for(std::chrono::milliseconds(500));
            }
        }
    );

    producer.join();
    consumer1.join();
    consumer2.join();
}
```

比如简单的stack
```cpp
int main(){
    std::stack<int> istack;
    istack.push(1);
    istack.push(2);
    std::cout << istack.top() << std::endl;  //2

    std::queue<int> iq;
    iq.push(1);
    iq.push(2);
    std::cout << iq.front() << std::endl;  //1 
    //队列没有top函数,只有front
    
}
```

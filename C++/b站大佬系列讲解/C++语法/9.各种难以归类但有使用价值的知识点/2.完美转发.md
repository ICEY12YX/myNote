C++完美转发的定义：完美转发是什么呢？说到底，<mark style="background: #ABF7F7A6;">它描述的其实就是一个参数传递的过程，能够将一个传递到一个函数的参数，再通过该函数原封不动的传递给另一个函数</mark>（这里的原封不动不单是指参数的值，更包括参数的类型，参数的限定符）
    需和上一课的万能引用一起用
```cpp
using boost::typeindex::type_id_with_cvr;

template<typename T>
void test2(T&& parm);

template<typename T>
void test1(T&& parm) {
	test2(parm);  //test1转发给test2
}

template<typename T>
void test2(T&& parm) {
	std::cout << type_id_with_cvr<decltype(parm)>().pretty_name() << std::endl;
}

int main() {
	int i = 10;
	test1(i);  //传入左值
	test1(10);  //传入右值

	//此时发现,不管传入左值还是右值,到test2了以后就都是左值引用了
}
```

所以用到std::forward
```cpp
using boost::typeindex::type_id_with_cvr;

template<typename T>
void test2(T&& parm);

template<typename T>
void test1(T&& parm) {
	test2(std::forward<T>(parm));  //必须写<T>
}

template<typename T>
void test2(T&& parm) {
	std::cout << type_id_with_cvr<decltype(parm)>().pretty_name() << std::endl;
}

int main() {
	int i = 10;
	test1(i);  //传入左值
	test1(10);  //传入右值
}
```

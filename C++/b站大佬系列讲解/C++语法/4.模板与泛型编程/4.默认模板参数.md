<mark style="background: #ABF7F7A6;">函数模板使用默认模板参数</mark>
```cpp
namespace mystd
{
	using default_func_type = std::function<void(int&)>;
	template<typename iter_type, typename func_type = default_func_type>  //给函数类型一个默认的类型值

	//上面给func_type一个默认的类型值,那使用func_type的时候,也要给一个默认的值
	void for_each(iter_type first, iter_type last, func_type func = [](int& elem) {
		++elem;
		})
	{
		for (auto iter = first; iter != last; ++iter) {
			func(*iter);
		}
	}
}

int main() {
	std::vector<int> ivec{ 1,2,3,4 };
	mystd::for_each(ivec.begin(), ivec.end());  //func不写了,即使用默认值
	for (int elem : ivec) {
		std::cout << elem << std::endl;
	}
}
```

<mark style="background: #ABF7F7A6;">类模板使用默认模板参数</mark>
```cpp
template<typename T, typename allocator = std::allocator<T>> //类模版头
class MyVector {
public:
	template<typename T2>  //成员函数模版头
	void outPut(const T2& elem)
	{
		std::cout << elem << std::endl;
	}
};

//写在外面
//template<typename T,typename allocator>
//template<typename T2>
//void MyVector<T,allocator>::outPut<T2>(const T2& elem) {  
//	std::cout << elem << std::endl;
//}

int main() {
	MyVector<int> myVec;
	myVec.outPut<int>(20);
}
```

<mark style="background: #ABF7F7A6;">模板参数就和普通函数的默认参数一样，一旦一个参数有了默认参数，它之后的参数都必须有默认参数</mark>

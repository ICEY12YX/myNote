死锁举例
```cpp
unsigned g_num = 0;
std::mutex myMutex1;
std::mutex myMutex2;

void test() {
	for (int i = 0; i < 1000000; i++) {
		std::lock_guard<std::mutex> lg1(myMutex1);  //应该在这里加锁,就是应该给尽量少的代码加锁
		//可能发生的情况是
		//1.这里lg1锁上了
		//3.想锁lg2,但是被主线程先锁了,等主线程解锁lg2
		std::lock_guard<std::mutex> lg2(myMutex2);  //这里为了演示加了两个锁
		++g_num;
	}
}

int main() {
	std::thread myThread(test);

	for (int i = 0; i < 1000000; i++) {
		std::lock_guard<std::mutex> lg2(myMutex2);
		//2.这里lg2锁上了
		//4.等子线程解锁lg1
		std::lock_guard<std::mutex> lg1(myMutex1);
		++g_num;
	}

	myThread.join();
	std::cout << g_num << std::endl;
}
```

解决办法:
<mark style="background: #ABF7F7A6;">只要让两个锁顺序一致就可以了</mark>
```cpp
unsigned g_num = 0;
std::mutex myMutex1;
std::mutex myMutex2;

void test() {
	for (int i = 0; i < 1000000; i++) {
		std::lock_guard<std::mutex> lg1(myMutex1);  //应该在这里加锁,就是应该给尽量少的代码加锁
		std::lock_guard<std::mutex> lg2(myMutex2);  //这里为了演示加了两个锁
		++g_num;
	}
}

int main() {
	std::thread myThread(test);

	for (int i = 0; i < 1000000; i++) {
		std::lock_guard<std::mutex> lg1(myMutex1);
		std::lock_guard<std::mutex> lg2(myMutex2);
		++g_num;
	}

	myThread.join();
	std::cout << g_num << std::endl;
}
```
(lg1,lg2的顺序无所谓,只要保证myMutex1,myMutex2顺序一样就好了)

<mark style="background: #ABF7F7A6;">但是让两个锁顺序一致常常是说起来容易，做起来难。</mark>于是C++11提供了<mark style="background: #ABF7F7A6;">std::lock</mark>。这个模板可以保证多个互斥锁绝对不会出现死锁的问题。同时提供了<mark style="background: #ABF7F7A6;">std::adopt_lock</mark>的功能来避免忘记释放锁的问题
    只要用了std::lock,就永远不会发生死锁问题
```cpp
void test() {
	for (int i = 0; i < 1000000; i++) {
		std::lock(myMutex1, myMutex2); //就这样写,同时加了两个锁
		++g_num;
	}
}
```
但是使用std::lock需要手动unlock,所以可以配合std::adopt_lock
```cpp
void test() {
	for (int i = 0; i < 1000000; i++) {
		std::lock(myMutex1, myMutex2);
		//加上std::adopt_lock,就是此时这个lg1只用关锁功能没有加锁功能
		std::lock_guard<std::mutex> lg1(myMutex1, std::adopt_lock); 
		std::lock_guard<std::mutex> lg2(myMutex2, std::adopt_lock);
		++g_num;
	}
}
```

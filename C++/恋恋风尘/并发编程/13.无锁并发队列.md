# 环形队列

要实现无锁并发，经常会用到一种结构无锁队列
它采用的是环状的队列结构
好处  一个是成环的队列大小是固定的，另外一个是我们通过移动头和尾就能实现数据的插入和取出

# 用锁实现环形队列

```cpp
template<typename T, size_t Cap>
class CircularQueLk :private std::allocator<T> {  //继承allocator
public:
    //_max_size为队列的总长度  cap是理论上放最多多少元素
    //我们上面知道
	CircularQueLk() :_max_size(Cap + 1), _data(std::allocator<T>::allocate(_max_size)), _head(0), _tail(0) {}

    //删掉拷贝赋值  拷贝构造
	CircularQueLk(const CircularQueLk&) = delete;
	CircularQueLk& operator = (const CircularQueLk&) volatile = delete;
	CircularQueLk& operator = (const CircularQueLk&) = delete;

    //析构
	~CircularQueLk() {
		//销毁前先加锁
		std::lock_guard<std::mutex> lock(_mtx);
		//循环销毁
		//调用内部元素的析构函数
		while (_head != _tail) {
    		//调用每个元素的析构(利用head的移动)
			std::allocator<T>::destroy(_data + _head);
			_head = (_head + 1) % _max_size;
		}
		//释放掉整个容器内存,要传容器大小
		std::allocator<T>::deallocate(_data, _max_size);
	}

	//先实现一个可变参数列表版本的插入函数做为基准函数
	template <typename ...Args>
	bool emplace(Args && ... args) {
		std::lock_guard<std::mutex> lock(_mtx);
		//判断队列是否满了,满了就不能再放入元素
		if ((_tail + 1) % _max_size == _head) {
			std::cout << "circular que full ! " << std::endl;
			return false;
		}
		//在尾部位置构造一个T类型的对象，构造参数为args...
		//这里才是构造出对象,上面的allocate只是分配那么大的内存
		std::allocator<T>::construct(_data + _tail, std::forward<Args>(args)...);
		//更新尾部元素位置
		_tail = (_tail + 1) % _max_size;
		return true;
	}

	//push 实现两个版本，一个接受左值引用，一个接受右值引用

	//接受左值引用版本
	bool push(const T& val) {
		std::cout << "called push const T& version" << std::endl;
		return emplace(val);  //可能会触发T类型的拷贝构造
	}

	// 接受右值引用版本，当然也可以接受左值引用，T&&为万能引用
	// 但是因为我们实现了const T&
	bool push(T&& val) {
		std::cout << "called push T&& version" << std::endl;
		return emplace(std::move(val));  //触发T类型的移动构造
	}

	//出队函数
	bool pop(T& val) {
		std::lock_guard<std::mutex> lock(_mtx);
		//判断头部和尾部指针是否重合，如果重合则队列为空
		if (_head == _tail) {
			std::cout << "circular que empty ! " << std::endl;
			return false;
		}
		//取出头部指针指向的数据
		val = std::move(_data[_head]);  //也可以不用move 单纯取个值
		//更新头部指针
		_head = (_head + 1) % _max_size;
		return true;
	}
private:
	size_t _max_size; //队列的总长度
	T* _data; //队列的
	std::mutex _mtx;  
	size_t _head = 0;
	size_t _tail = 0;
};
```

# 无锁队列

如果我们用原子变量而不是用锁实现环形队列，那就是无锁并发的队列了

之前提到的原子变量的读改写操作
```cpp
bool std::atomic<T>::compare_exchange_weak(T &expected, T desired);
bool std::atomic<T>::compare_exchange_strong(T &expected, T desired);
```
`compare_exchange_strong`会比较原子变量`atomic<T>`的值和`expected`的值是否相等，
如果相等则执行交换操作，
将`atomic<T>`的值换为`desired`并且返回true,
否则将expected的值修改为原子变量`atomic<T>`的值，并且返回false.

`compare_exchange_weak`功能比`compare_exchange_strong`弱一些，
他不能保证`atomic<T>`的值和`expected`的值相等时也会做交换，
很可能原子变量和预期值相等也会返回false，所以使用要多次循环使用。

伪代码
```cpp
template <typename T>
bool atomic<T>::compare_exchange_strong(T &expected, T desired) {
    std::lock_guard<std::mutex> guard(m_lock);
    if (m_val == expected)
        return m_val = desired, true;
    else
        return expected = m_val, false;
}
```

`compare_exchange_weak`功能比`compare_exchange_strong`弱一些，
他不能保证`atomic<T>`的值和`expected`的值相等时也会做交换，
很可能原子变量和预期值相等也会返回false，所以使用要多次循环使用。

下面代码
只不过将类的成员变量mutex换成atomic类型的原子变量, 
可以利用自旋锁的思路  将锁  替换为  原子变量循环检测，进而达到锁住互斥逻辑的效果
```cpp
template<typename T, size_t Cap>
class CircularQueSeq :private std::allocator<T> {
public:
    //和上面差不多
	CircularQueSeq() :_max_size(Cap + 1), _data(std::allocator<T>::allocate(_max_size)), _atomic_using(false), _head(0), _tail(0) {}
	CircularQueSeq(const CircularQueSeq&) = delete;
	CircularQueSeq& operator = (const CircularQueSeq&) volatile = delete;
	CircularQueSeq& operator = (const CircularQueSeq&) = delete;

	~CircularQueSeq() {
		//循环销毁
		//调用内部元素的析构函数
		while (_head != _tail) {
			std::allocator<T>::destroy(_data + _head);
			_head = (_head + 1) % _max_size;
		}
		//调用回收操作
		std::allocator<T>::deallocate(_data, _max_size);
	}

	//先实现一个可变参数列表版本的插入函数最为基准函数
	template <typename ...Args>
	bool emplace(Args && ... args) {

        //就多了这一段  "上锁"
		bool use_expected = false;
		bool use_desired = true;
		do
		{
			use_expected = false;
			use_desired = true;
		} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));

		//判断队列是否满了
		if ((_tail + 1) % _max_size == _head) {
			std::cout << "circular que full ! " << std::endl;

            //"解锁"
			do
			{
				use_expected = true;
				use_desired = false;
			} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));
			return false;
		}
		std::allocator<T>::construct(_data + _tail, std::forward<Args>(args)...);
		_tail = (_tail + 1) % _max_size;

        //"解锁"
		do
		{
			use_expected = true;
			use_desired = false;
		} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));

		return true;
	}

	//push 实现两个版本，一个接受左值引用，一个接受右值引用

	//接受左值引用版本
	bool push(const T& val) {
		std::cout << "called push const T& version" << std::endl;
		return emplace(val);
	}

	//接受右值引用版本，当然也可以接受左值引用，T&&为万能引用
	// 但是因为我们实现了const T&
	bool push(T&& val) {
		std::cout << "called push T&& version" << std::endl;
		return emplace(std::move(val));
	}

	//出队函数
	bool pop(T& val) {

        //"上锁"
		bool use_expected = false;
		bool use_desired = true;
		do
		{
			use_desired = true;
			use_expected = false;
		} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));
		//判断头部和尾部指针是否重合，如果重合则队列为空
		if (_head == _tail) {
			std::cout << "circular que empty ! " << std::endl;

            //"解锁"
			do
			{
				use_expected = true;
				use_desired = false;
			} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));
			return false;
		}
		//取出头部指针指向的数据
		val = std::move(_data[_head]);
		//更新头部指针
		_head = (_head + 1) % _max_size;

        //"解锁"
		do
		{
			use_expected = true;
			use_desired = false;
		} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));
		return true;
	}
private:
	size_t _max_size;
	T* _data;
	std::atomic<bool> _atomic_using;  //多了一个原子变量
	size_t _head = 0;
	size_t _tail = 0;
};

```
上面代码最大的不同就是这一段
```cpp
bool use_expected = false;
bool use_desired = true;
do
{
	use_desired = true;
	use_expected = false;
} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));

//...

bool use_expected = false;
bool use_desired = true;
do
{
	use_desired = false;
	use_expected = true;
} while (!_atomic_using.compare_exchange_strong(use_expected, use_desired));
```
就是实现了一个自旋锁(虽然有点消耗性能  就是一个线程进不去就会循环等待)
初始_atomic_using=false
然后线程1 进来了  判断_atomic_using == use_expected
所以_atomic_using = use_desired 并返回true(非一下是false)  所以退出循环  等于抢到锁了

此时线程2 进来_atomic_using=true  那就不等于use_expected  
然后use_expected ==  _ atomic_using   返回false(之后循环use_expected会重新变回false)
所以就在那里卡着,循环等待

后面线程1要退出了
_ atomic_using == use_expected(这个时候是true)  然后_atomic_using = use_desired  
就_atomic_using变回false  线程1退出循环,解锁了  然后线程2就能上锁进去了

# 单一原子变量的弊端

多个线程push和pop操作耦合读太高,就是同一时刻仅有一个线程pop或者push
其实可以一个线程push的同时,另一个线程可以pop

所以可以上面代码改改
成员变量改为
```cpp
private:
	size_t _max_size;
	T* _data;
	std::atomic<size_t>  _head;
	std::atomic<size_t> _tail;
```

就看重点函数
```cpp
	//出队函数
	bool pop(T& val) {

		size_t h;
		do
		{
			//取出head
			h = _head.load();  //1 处
			//判断头部和尾部指针是否重合，如果重合则队列为空
			if (h == _tail.load())
			{
				return false;
			}
			//不为空,取出数据
			val = _data[h]; // 2处

		} while (!_head.compare_exchange_strong(h,
			(h + 1) % _max_size)); //3 处

		return true;
	}
```
上面那段do while相当于上了锁

解释
    ![[Pasted image 20250715232409.png]]
    ![[Pasted image 20250715232429.png]]
    ![[Pasted image 20250715232533.png]]
    之所以能这样,因为_head是原子的
    同时只有一个线程能成功执行`_head.compare_exchange_strong(h,(h + 1) % _max_size);`
    这句在互斥的同时, 让_ head+1了  
    线程抢占到资源后 正好_ head+1了  就直接pop下一个元素

注意上面`val = _data[h];` 不能用move  因为可能两个线程同时给val赋值同一个位置上的元素, 用move移动掉的话,后移动的线程会发现里面没东西好移动   会报错

push函数的逻辑乍一看和pop一样，但是我们会发现多线程push的情况存在线程安全问题
解释
    ![[Pasted image 20250715233838.png]]
    就是线程1先退出循环  但是实际看上去是线程2改动成功了

```cpp
    bool push(T& val)
    {
        size_t t;
        do
        {
            t = _tail.load(); //1
            //判断队列是否满
```

但是仍然有问题
是push和pop联动的问题
比如 有个之前已经删除的值,但是我们上面说过 删除不是真的删 而是等之后覆盖
线程1  先push  然后退出了循环  但是还没有修改队列里的值(这个位置还存着之前被pop出去的值,现在还没被覆盖)
然后线程2 来pop了  它pop了这个还没被覆盖的旧值
这就会出问题

所以要再改一下
加入一个新的原子变量_tail_update  用来标记尾部是否push成功
只有数据真的被push进去 _ tail才和_ tail_update 相等
```cpp
    bool push(const T& val)
    {
        size_t t;
        do
        {
            t = _tail.load();  //1
            //判断队列是否满
            if( (t+1)%_max_size == _head.load())
            {
                return false;
            }

        } while (!_tail.compare_exchange_strong(t,
            (t + 1) % _max_size));  //3

        _data[t] = val; //2
        
        size_t tailup;
        do
        {
            tailup = t;  //把tailup设置成push以前tail的位置

        } while (_tail_update.compare_exchange_strong(tailup, 
            (tailup + 1) % _max_size));
        //如果_tail_update 等于 tailup 
        //那_tail_update 就等于 tailup+1(此时_tail的值)
        //然后返回true 继续循环
        //然后再次循环 就会返回false  退出循环
        return true;
    }

    bool pop(T& val) {

        size_t h;
        do
        {
            h = _head.load();  //1 处
            //判断头部和尾部指针是否重合，如果重合则队列为空
            if(h == _tail.load())
            {
                return false;
            }

            //判断如果此时要读取的数据和tail_update是否一致，
            //如果一致说明尾部数据未更新完,返回false,让pop失败先
            if(h == _tail_update.load())
            {
                return false;
            }
            val = _data[h]; // 2处

        } while (!_head.compare_exchange_strong(h, 
            (h+1)% _max_size)); //3 处

        return true;
    }
```
push的下面那个修改_tail_update的循环  线程修改成功值以后  
第一次执行这个循环 虽然_tail_update 就等于 tailup+1(此时_tail的值)了   
但是因为返回true  所以肯定再循环一次,才退出
总之  意思是  如果_tail_update还没改成最新的值 那就修改下    如果_tail_update不是旧值了  才让过

# 优化性能

因为我们知道原子操作默认采用的是`memory_order_seq_cst`内存顺序，性能上不是最优的，我们可以用`acquire`和`release`的内存顺序实现同步的效果。
```cpp
    bool pop(T& val) {

        size_t h;
        do
        {
            h = _head.load(std::memory_order_relaxed);  //1 处
            //判断头部和尾部指针是否重合，如果重合则队列为空
            //acquire _tail
            if (h == _tail.load(std::memory_order_acquire)) //2处
            {
                std::cout << "circular que empty ! " << std::endl;
                return false;
            }

            //判断如果此时要读取的数据和tail_update是否一致，如果一致说明尾部数据未更新完
            //acquire _tail_update
            if (h == _tail_update.load(std::memory_order_acquire)) //3处
            {
                return false;
            }
            val = _data[h]; // 2处

        } while (!_head.compare_exchange_strong(h,
            (h + 1) % _max_size, std::memory_order_release, 
            std::memory_order_relaxed)); //4 处
        //意思是_head被成功修改的时候 用release模式  没修改(就是false)的时候用relaxed模式
        //这样节省性能
        
        std::cout << "pop data success, data is " << val << std::endl;
        return true;
    }
```
上面两处用acquire, 其实是配合下面的push(对_tail和_tail_update修改的时候都用release模式)
就是_tail和_tail_update被读取的时候  之前别的线程对_tail和_tail_update的修改 能被要读_tail和_tail_update的线程看到
总之 就是release和acquire相配合  
release保证这之前的修改能被看见
acquire保证这之后的操作能看到release之前的那些修改

同理_head的读写也用release和acquire

<mark style="background: #FFF3A3A6;">1 处为`memory_order_relaxed`是因为即使多个线程pop，每个线程获取的head可能不及时，这个没关系，因为我们有4处的while来重试</mark>
就是h拿到旧值也没事  后面_ head和h比较的时候用的是release(此时_ head肯定是新值),如果是旧值 那就再循环来一次就好了  用relaxed可以节省性能

```cpp
    bool push(const T& val)
    {
        size_t t;
        do
        {
            t = _tail.load(std::memory_order_relaxed);  //5
            //判断队列是否满
            if ((t + 1) % _max_size == _head.load(std::memory_order_acquire))
            {
                std::cout << "circular que full ! " << std::endl;
                return false;
            }

        } while (!_tail.compare_exchange_strong(t,
            (t + 1) % _max_size, std::memory_order_release, 
            std::memory_order_relaxed));  //6

        _data[t] = val; 
        size_t tailup;
        do
        {
            tailup = t;

        } while (_tail_update.compare_exchange_strong(tailup,
            (tailup + 1) % _max_size, std::memory_order_release, std::memory_order_relaxed)); //7

        std::cout << "called push data success " << val << std::endl;
        return true;
    }
```

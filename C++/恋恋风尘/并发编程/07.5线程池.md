```cpp
#ifndef __THREAD_POOL_H__
#define __THREAD_POOL_H__

#include <atomic>
#include <condition_variable>
#include <future>
#include <iostream>
#include <mutex>
#include <queue>
#include <thread>
#include <vector>

class ThreadPool  {
public:
    //拷贝构造,拷贝赋值删了
    ThreadPool(const ThreadPool&) = delete;
    ThreadPool& operator=(const ThreadPool&) = delete;

    //实现单例,保证线程池(这一整个池)是唯一的
    //不是线程是唯一的
    static ThreadPool& instance() {
        static ThreadPool ins;
        return ins;
    }

    //using就是typedef
    //就是定义了外部传过来的任务类型都是std::packaged_task<void()>;
    //然后有个简短的 Task来定义
    using Task = std::packaged_task<void()>;

    //析构
    ~ThreadPool() {
        stop();
    }

    //commit  投递任务进任务队列
    //F为回调函数  ... Args是可变参数(因为不知道前面的F回调函数里面要传几个参数)
    template <typename F, typename... Args>
    //用到万能引用
    //用到decltype(f(args...))尾置类型推导,推出回调函数要返回什么类型,然后给std::future
    //(commit函数返回的类型是std::future<decltype(f(args...)))
    auto commit(F&& f, Args&&... args) -> std::future<decltype(f(args...))> {
        //用别名存  回调函数要返回的类型
        using RetType = decltype(f(args...));

        //stop_.load()就是取现在stop_是true还是false
        //如果是true,说明线程池要关了
        if (stop_.load())
            return std::future<RetType>{};  //返回空的future,然后外部再去做处理

        //是false,执行下面的
        auto task = std::make_shared<std::packaged_task<RetType()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...));

        //从task任务中返回一个future
        //(看看前面的,std::packaged_task也可以用get_future返回future)
        //因为上面task是指针了,所以用->调用get_future函数
        std::future<RetType> ret = task->get_future();
        {
            std::lock_guard<std::mutex> cv_mt(cv_mt_);  //锁上
            tasks_.emplace([task] { (*task)(); });  //加入任务队列
        }
        cv_lock_.notify_one();  
        return ret;  //返回上面的future(这样就可以在外部拿到这个task执行的结果)
    }

    int idleThreadCount() {
        return thread_num_;
    }

private:
    //因为线程池是单例的,所以构造函数需要时私有的
	ThreadPool(unsigned int num = 5) :stop_(false) {  //默认构造函数
		{
			if (num < 1) {
				thread_num_ = 1;  //最少线程数为1
			}
			else {
				thread_num_ = num;  //或者外面指定线程数(或者上面默认是5个线程)
			}
		}//这里有个作用域(暂时不知道为什么这样写)
		start();
	}

    //启动线程池
    void start() {
        //循环thread_num_次
        //每次往pool_数组里扔进去线程
        //线程里面执行我们下面写好的回调(这个回调暂时不知道在做什么)
        for (int i = 0; i < thread_num_; ++i) {  
            pool_.emplace_back([this]() {
                while (!this->stop_.load()) {
                    Task task;
                    {
                        std::unique_lock<std::mutex> cv_mt(cv_mt_);
                        this->cv_lock_.wait(cv_mt, [this] {
                            return this->stop_.load() || !this->tasks_.empty();
                        });
                        if (this->tasks_.empty())
                            return;

                        task = std::move(this->tasks_.front());
                        this->tasks_.pop();
                    }
                    this->thread_num_--;
                    task();
                    this->thread_num_++;
                }
            });
        }
    }

    //结束所有进程
    void stop() {
        stop_.store(true);
        cv_lock_.notify_all();  //通知所有线程?
        for (auto& td : pool_) {  //遍历所有线程
            if (td.joinable()) {
                std::cout << "join thread " << td.get_id() << std::endl;
                td.join();  //等所有线程执行结束
            }
        }
    }

private:
    std::mutex               cv_mt_;  //锁
    std::condition_variable  cv_lock_;  //条件变量
    std::atomic_bool         stop_;  //控制停止条件
    std::atomic_int          thread_num_;  //线程数量
    std::queue<Task>         tasks_;  //任务队列
    std::vector<std::thread> pool_;  //用vector存储线程队列
};

#endif  // !__THREAD_POOL_H__
```

线程池什么时候不建议用
1. 任务之间有时间顺序要求(线程池的线程都是并行执行,保证不了顺序,要么就把几个有顺序的任务写成一个线程)
2. 任务之间有逻辑关联,比如都要访问某个互斥的资源,这样不能保证哪个线程先访问到互斥资源

# 散列表

我们自己在设计散列函数的函数时应该遵循什么规则呢？
1. 得到的散列值是一个非负整数
2. 两个相同的键，通过散列函数计算出的散列值也相同
3. 两个不同的键，计算出的散列值不同(但是肯定达不到,会存在冲突,就用下面的链表法解决)

# 设计思路
我们要实现上述逻辑，可以考虑将11，12，13等hash值放入一个vector中。

多线程根据key计算得出hash值的过程并不需要加锁，可以实现并行计算。
但是对于链表的增删改查需要加锁。

所以我们考虑将链表封装为一个类`bucket_type`,支持数据的增删改查。
将整体的查找表封装为`threadsafe_lookup_table`类，实现散列规则和调度`bucket_type`类。

```cpp
    template<typename Key, typename Value, typename Hash = std::hash<Key>>
    typedef std::pair<Key, Value> bucket_value;
    typedef std::list<bucket_value> bucket_data;
    typedef typename bucket_data::iterator bucket_iterator;
    //为了对照起来看

    //查找key值，返回对应的value，未找到则返回默认值
    //(这里这个默认值是我们传进来的)
    Value value_for(Key const& key, Value const& default_value) 
    {
        //查之前加共享锁
        //这样别的线程可以一起来查加共享锁,但是写线程加不上锁
        std::shared_lock<std::shared_mutex> lock(mutex);
        //返回key对应位置的迭代器
        bucket_iterator const found_entry = find_entry_for(key);
        //found_entry == data.end()说明没找到,那就返回默认值
        return (found_entry == data.end()) ?
                default_value : found_entry->second;
    }
    
    //添加key和value，找到则更新，没找到则添加
    void add_or_update_mapping(Key const& key, Value const& value)
    {
        //先加锁
        std::unique_lock<std::shared_mutex> lock(mutex);
        //找到对应key的迭代器
        bucket_iterator const found_entry = find_entry_for(key);
        if (found_entry == data.end())  
        {
            //没找到就添加
            //bucket_value是pair类型,
            data.push_back(bucket_value(key, value));
        }
        else
        {
            //找到了,就去更新值为我们传进来的值
            //因为key和value是一一对应的
            //所以此时必定要修改value
            found_entry->second = value;
        }
    }
    //删除对应的key
    void remove_mapping(Key const& key)
    {
        // //先加锁
        std::unique_lock<std::shared_mutex> lock(mutex);
        //找到对应key的迭代器
        bucket_iterator const found_entry = find_entry_for(key);
        
        if (found_entry != data.end())
        {
            //找到了,就擦除
            data.erase(found_entry);
        }
    }
```

再实现`threadsafe_lookup_table`类
查找表的结构
```cpp
    template<typename Key, typename Value, typename Hash = std::hash<Key>>
    typedef std::pair<Key, Value> bucket_value;
    typedef std::list<bucket_value> bucket_data;
    typedef typename bucket_data::iterator bucket_iterator;
    class bucket_type
    //为了对照起来看

class threadsafe_lookup_table{
private:
    //用vector存储桶类型
    std::vector<std::unique_ptr<bucket_type>> buckets;
    //hash<Key> 哈希表 
    //上面我们模版定义了Hash = std::hash<Key>>
    Hash hasher;

    //根据key生成数字，并对桶的大小取余得到下标，根据下标返回对应的桶智能指针
    bucket_type& get_bucket(Key const& key) const
    {
        //hasher(key)会返回一个哈希值(因为key有可能是字符串之类的,所以需要一个数值的哈希值)
        //哈希值和桶大小取余,得到我们到底应该存哪里
        std::size_t const bucket_index = hasher(key) % buckets.size();
        //buckets数组里存的是unique_ptr,所以*取内容
        return *buckets[bucket_index];
    }
};
```

public方法
```cpp
    //默认桶大小是19
    threadsafe_lookup_table(
        unsigned num_buckets = 19, Hash const& hasher_ = Hash()) :
        buckets(num_buckets), hasher(hasher_)
    {
        for (unsigned i = 0; i < num_buckets; ++i)
        {
            //给buckets数组每个元素进行初始化
            //那就形成了一个vector叫buckets
            //然后里面的元素是bucket_type
            //bucket_type里面又有一个list类型的bucket_data data
            buckets[i].reset(new bucket_type);
        }
    }

    threadsafe_lookup_table(threadsafe_lookup_table const& other) = delete;
    threadsafe_lookup_table& operator=(
        threadsafe_lookup_table const& other) = delete;

    //查找表
    //Value()就是 Value反正是模版  Value()就是调用对应类型构造函数,生成默认值
    Value value_for(Key const& key,
        Value const& default_value = Value()) 
    {
        //get_bucket找key应该存在vector数组中的哪个位置
        //(返回对应位置的bucket_type类型)就是相当于返回那条链表
        //value_for是bucket_type类型的方法
        //value_for查找key值，返回对应的value，未找到则返回默认值
        return get_bucket(key).value_for(key, default_value);
    }

    //添加元素
    void add_or_update_mapping(Key const& key, Value const& value)
    {
        //同样调用bucket_type类型的方法
        //add_or_update_mapping 添加key和value，找到则更新，没找到则添加
        get_bucket(key).add_or_update_mapping(key, value);
    }

    //删除
    void remove_mapping(Key const& key)
    {
        //remove_mapping删除对应的key
        get_bucket(key).remove_mapping(key);
    }
```

可将当前查找表的副本作为一个map返回
```cpp
std::map<Key, Value> get_map() 
{
    //创建一个存锁的数组,每个锁是unique_lock,同时每个锁用的是shared_mutex(共享锁)
    std::vector<std::unique_lock<std::shared_mutex>> locks;
    for (unsigned i = 0; i < buckets.size(); ++i)
    {
        //buckets是数组
        //buckets[i]->mutex得到bucket_type里的mutex
        //就是对整个表的每条链上锁
        locks.push_back(
                std::unique_lock<std::shared_mutex>(buckets[i]->mutex));
    }

    //创建一个std::map类型的res
    std::map<Key, Value> res;
    for (unsigned i = 0; i < buckets.size(); ++i)
    {
        //得到每个bucket_type中的链表中的初始位置迭代器
        //需用typename告诉编译器bucket_type::bucket_iterator是一个类型，以后再实例化
        //当然此处可简写成auto it = buckets[i]->data.begin();
        typename bucket_type::bucket_iterator it = buckets[i]->data.begin();
        for (;it != buckets[i]->data.end();++it)
        {
            //把每条链表里的pair存进res
            //*it就是取出data链表中的每个数据(是std::pair<Key, Value>)
            res.insert(*it);
        }
    }
    return res;
}
```

完整代码
```cpp
#pragma once
#include <iostream>

#include <thread>
#include <vector>
#include <list>
#include <memory>
#include <mutex>
#include <shared_mutex>
#include <iterator>
#include <map>

template<typename Key, typename Value, typename Hash = std::hash<Key>>
class threadsafe_lookup_table
{
private:

	// 桶类型
	class bucket_type //类可以嵌套的
	{
		friend class threadsafe_lookup_table;
	private:
        //存储元素的类型为pair，由key和value构成
        typedef std::pair<Key, Value> bucket_value;
        //由链表存储元素构(链表就用系统自给的)
        typedef std::list<bucket_value> bucket_data;
        //链表的迭代器
        typedef typename bucket_data::iterator bucket_iterator;

        //链表数据
        bucket_data data; //(data是整个链表)
        //改用共享锁(可以实现读写锁，保证读的时候可以并发读。)
        mutable std::shared_mutex mutex;

		//查找操作，在list中找到匹配的key值，然后返回迭代器
		bucket_iterator find_entry_for(const Key & key)
		{
			return std::find_if(data.begin(), data.end(),
				[&](bucket_value const& item)
				{return item.first == key; });
		}
	
	public:
		//查找key值，找到返回对应的value，未找到则返回默认值
		Value value_for(Key const& key, Value const& default_value) 
		{
			std::shared_lock<std::shared_mutex> lock(mutex);
			bucket_iterator const found_entry = find_entry_for(key);
			return (found_entry == data.end()) ?
				default_value : found_entry->second;
		}
		//添加key和value，找到则更新，没找到则添加
		void add_or_update_mapping(Key const& key, Value const& value)
		{
			std::unique_lock<std::shared_mutex> lock(mutex);
			bucket_iterator const found_entry = find_entry_for(key);
			if (found_entry == data.end())
			{
				data.push_back(bucket_value(key, value));
			}
			else
			{
				found_entry->second = value;
			}
		}
		//删除对应的key
		void remove_mapping(Key const& key)
		{
			std::unique_lock<std::shared_mutex> lock(mutex);
			bucket_iterator const found_entry = find_entry_for(key);
			if (found_entry != data.end())
			{
				data.erase(found_entry);
			}
		}
	};
	
	//用vector存储桶类型
	std::vector<std::unique_ptr<bucket_type>> buckets;
	//hash<Key> 哈希表 用来根据key生成哈希值
	Hash hasher;

	//根据key生成数字，并对桶的大小取余得到下标，根据下标返回对应的桶智能指针
	bucket_type& get_bucket(Key const& key) const
	{
		std::size_t const bucket_index = hasher(key) % buckets.size();
		return *buckets[bucket_index];
	}


public:

	threadsafe_lookup_table(
		unsigned num_buckets = 19, Hash const& hasher_ = Hash()) :
		buckets(num_buckets), hasher(hasher_)
	{
		for (unsigned i = 0; i < num_buckets; ++i)
		{
			buckets[i].reset(new bucket_type);
		}
	}

	threadsafe_lookup_table(threadsafe_lookup_table const& other) = delete;
	threadsafe_lookup_table& operator=(
		threadsafe_lookup_table const& other) = delete;

	Value value_for(Key const& key,
		Value const& default_value = Value()) 
	{
		return get_bucket(key).value_for(key, default_value);
	}

	void add_or_update_mapping(Key const& key, Value const& value)
	{
		get_bucket(key).add_or_update_mapping(key, value);
	}

	void remove_mapping(Key const& key)
	{
		get_bucket(key).remove_mapping(key);
	}

	std::map<Key, Value> get_map() 
	{
		std::vector<std::unique_lock<std::shared_mutex>> locks;
		for (unsigned i = 0; i < buckets.size(); ++i)
		{
			locks.push_back(
				std::unique_lock<std::shared_mutex>(buckets[i]->mutex));
		}
		std::map<Key, Value> res;
		for (unsigned i = 0; i < buckets.size(); ++i)
		{
			//需用typename告诉编译器bucket_type::bucket_iterator是一个类型，以后再实例化
			//当然此处可简写成auto it = buckets[i]->data.begin();
			typename bucket_type::bucket_iterator it = buckets[i]->data.begin();
			for (;it != buckets[i]->data.end();++it)
			{
				res.insert(*it);
			}
		}
		return res;
	}
};
```

```cpp
void TestThreadSafeHash() {
    std::set<int> removeSet;   //存储移除数据的key
    threadsafe_lookup_table<int, std::shared_ptr<MyClass>> table;
    
    std::thread t1([&]() {
    	for(int i = 0; i < 100; i++) {
        	//创建100个Myclass类的变量,并且用智能指针包裹
           auto class_ptr = std::make_shared<MyClass>(i); 
           //把数据插入table  i是key  class_ptr是value
           table.add_or_update_mapping(i, class_ptr);
    	}
    });

    std::thread t2([&]() {
        for (int i = 0; i < 100; ) {
            //查找key为i的数据
            auto find_res = table.value_for(i, nullptr);
            if(find_res) {
                //找到了,就移除
                table.remove_mapping(i);
                //将移除的key存入set
                removeSet.insert(i);
                i++;
            }

            std::this_thread::sleep_for(std::chrono::milliseconds(10));
        }
    });

    //和线程1做一样的事情
    std::thread t3([&]() {
        for (int i = 100; i < 200; i++) {
            auto class_ptr = std::make_shared<MyClass>(i);
            table.add_or_update_mapping(i, class_ptr);
        }
        });


    t1.join();
    t2.join();
    t3.join();

    for(auto & i : removeSet)
    {
        std::cout << "remove data is " << i << std::endl;
    }

	auto copy_map =  table.get_map();
    for(auto & i : copy_map)
    {
        std::cout << "copy data is " << *(i.second) << std::endl;
    }
}

int main()
{
    TestThreadSafeHash();
}
```

```python
def piecewise_linear_transform(image, points):
    """
    还看不懂,总之就是 按一定的规律划分区域,应用不同变换规则???
    分段线性变换
    points: [(输入值, 输出值), ...]
    例如：[(0,0), (100,30), (200,220), (255,255)]
    """
    # 创建查找表
    lut = np.zeros(256, dtype=np.uint8)
    
    # 对每一段进行线性插值
    for i in range(len(points)-1):
        x1, y1 = points[i]
        x2, y2 = points[i+1]
        
        # 计算这一段的变化斜率
        for x in range(x1, x2+1):
            ratio = (x - x1) / (x2 - x1) if x2 != x1 else 0
            lut[x] = np.clip(y1 + ratio * (y2 - y1), 0, 255)
    
    # 应用变换
    return lut[image]
```

**对比度展宽**
```python
def contrast_stretching(image, a=100, b=150, ga=50, gb=200):
    """
    还看不懂
    对比度展宽 - 让中间灰度范围更明显
    a, b: 输入灰度范围
    ga, gb: 输出灰度范围
    """
    # 创建查找表
    lut = np.zeros(256, dtype=np.uint8)
    
    # 三段变换
    for i in range(256):
        if i < a:
            lut[i] = (ga / a) * i  # 第一段 (大概是按比例来)
        elif i < b:
            lut[i] = ga + ((gb - ga) / (b - a)) * (i - a)  # 第二段（对比度增强）
        else:
            lut[i] = gb + ((255 - gb) / (255 - b)) * (i - b)  # 第三段
    
    return lut[image]
```
(让黑的,白的变更黑更白, 中间一小部分灰色"拉宽",也变黑一点或白一点)


**灰度级切片**
```python
def gray_level_slicing(image, low=100, high=150, background=0):
    """
    灰度级切片 - 突出显示特定灰度范围
    low, high: 关注的灰度范围
    background: 背景灰度值
    """
    result = image.copy()
    # 将不在关注范围内的像素设为背景
    result[(image < low) | (image > high)] = background
    return result
```
